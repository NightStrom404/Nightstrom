import os
import json
import time
import requests
from urllib.parse import urlparse

SCANREPO_API = "https://www.scanrepo.dev/api/scan"
USER_AGENT = "ScanRepoCLI/1.0"

def clear_screen():
    os.system("cls" if os.name == "nt" else "clear")

def banner():
    print(r"""
   _____                       _____
  / ____|                     |  __ \\
 | (___   ___ __ _ _ __       | |__) |___ _ __   ___
  \\___ \\ / __/ _` | '_ \\      |  _  // _ \\ '_ \\ / _ \\

                                          | |
                                          |_|

                 GITHUB REPO SCANNER
              Security & Code Analysis CLI
""")

def status(message, level="*"):
    now = time.strftime("%H:%M:%S")
    print(f"[{now}] [{level}] {message}")

def valid_github_url(url):
    try:
        parsed = urlparse(url)
        if parsed.scheme not in ("http", "https"):
            return False
        if parsed.netloc.lower() not in ("github.com", "www.github.com"):
            return False
        path = parsed.path.strip("/")
        return len(path.split("/")) >= 2
    except Exception:
        return False

def get_repo_url():
    print("  Masukkan URL repository GitHub")
    print("  Contoh: https://github.com/user/repository")
    print()
    while True:
        url = input("  GitHub URL > ").strip()
        if not url:
            print("  [!] URL tidak boleh kosong.\\n")
            continue
        if not valid_github_url(url):
            print("  [!] URL GitHub tidak valid.\\n")
            continue
        return url

def start_scan(repo_url):
    headers = {
        "User-Agent": USER_AGENT,
        "Accept": "application/json",
        "Content-Type": "application/json"
    }
    status("Mengirim scan request...")
    status(f"Repository : {repo_url}")
    try:
        response = requests.post(
            SCANREPO_API,
            json={"url": repo_url, "force": False},
            headers=headers,
            timeout=30
        )
        status(f"HTTP Status: {response.status_code}", "*" if response.ok else "!")
        if not response.ok:
            return None, None
        scan_id = None
        result = None
        for line in response.text.strip().splitlines():
            if not line.strip():
                continue
            try:
                data = json.loads(line)
            except json.JSONDecodeError:
                continue
            if data.get("type") == "progress":
                step = data.get("step", "unknown")
                progress = data.get("progress", 0)
                status(f"{step} - {progress}%")
            elif data.get("type") == "result":
                result = data.get("data", {})
                scan_id = result.get("commitSha")
                status("Scan selesai.", "+")
        return scan_id, result
    except requests.Timeout:
        status("Request timeout.", "!")
        return None, None
    except requests.ConnectionError:
        status("Tidak dapat terhubung ke ScanRepo.", "!")
        return None, None
    except requests.RequestException as error:
        status(f"Request error: {error}", "!")
        return None, None
    except Exception as error:
        status(f"Error: {error}", "!")
        return None, None

def wrap_text(text, width=52):
    if not text:
        return ["-"]
    words = str(text).split()
    lines = []
    current = ""
    for word in words:
        if len(current) + len(word) + 1 > width:
            lines.append(current)
            current = word
        else:
            current += (" " if current else "") + word
    if current:
        lines.append(current)
    return lines

def clean_snippet(snippet):
    if not snippet:
        return None
    snippet = str(snippet).replace("\\n", " ").replace("\\r", " ").replace("\\t", " ")
    snippet = " ".join(snippet.split())
    if len(snippet) > 150:
        snippet = snippet[:150] + "..."
    return snippet

def show_finding(number, finding):
    title = finding.get("title", "Unknown finding")
    severity = finding.get("severity", "unknown")
    category = finding.get("category", "unknown")
    rule_id = finding.get("ruleId", "unknown")
    points = finding.get("points", 0)
    file_path = finding.get("filePath") or "-"
    description = finding.get("description", "No description available.")
    snippet = clean_snippet(finding.get("snippet"))
    print()
    print("  ┌" + "─" * 58 + "┐")
    print(f"  │ FINDING #{number:<47} │")
    print("  ├" + "─" * 58 + "┤")
    print(f"  │ Title    : {title[:47]:<47} │")
    print(f"  │ Severity : {severity:<47} │")
    print(f"  │ Category : {category:<47} │")
    print(f"  │ Rule ID  : {rule_id:<47} │")
    print(f"  │ Points   : {str(points):<47} │")
    print("  │ File     :")
    for line in wrap_text(file_path):
        print(f"  │   {line:<54}│")
    print("  │")
    print("  │ Description:")
    for line in wrap_text(description):
        print(f"  │   {line:<54}│")
    if snippet:
        print("  │")
        print("  │ Snippet:")
        for line in wrap_text(snippet):
            print(f"  │   {line:<54}│")
    print("  └" + "─" * 58 + "┘")

def show_findings(findings):
    print()
    print("┌" + "─" * 60 + "┐")
    print("│ FINDINGS" + " " * 51 + "│")
    print("└" + "─" * 60 + "┘")
    if not findings:
        print()
        print("  Tidak ada findings.")
        return
    for number, finding in enumerate(findings, 1):
        show_finding(number, finding)

def show_badges(badges):
    if not badges:
        return
    print()
    print("┌" + "─" * 60 + "┐")
    print("│ REPOSITORY INFO" + " " * 44 + "│")
    print("└" + "─" * 60 + "┘")
    for badge in badges:
        label = badge.get("label", "unknown")
        description = badge.get("description", "")
        print()
        print(f"  • {label}")
        for line in wrap_text(description, 56):
            print(f"    {line}")

def show_result(result):
    if not result:
        status("Tidak ada data hasil scan.", "!")
        return
    meta = result.get("meta", {})
    repository = meta.get("repo", "unknown")
    language = meta.get("language", "unknown")
    risk_score = result.get("riskScore", 0)
    risk_level = result.get("riskLevel", "unknown")
    findings = result.get("findings", [])
    badges = result.get("badges", [])
    print()
    print("╔══════════════════════════════════════════════════════════════╗")
    print("║                       SCAN RESULT                            ║")
    print("╚══════════════════════════════════════════════════════════════╝")
    print()
    print(f"  Repository : {repository}")
    print(f"  Language   : {language}")
    print(f"  Risk Score : {risk_score}")
    print(f"  Risk Level : {risk_level}")
    print(f"  Findings   : {len(findings)}")
    show_findings(findings)
    show_badges(badges)
    print()
    print("═" * 62)

def main():
    clear_screen()
    banner()
    repo_url = get_repo_url()
    print()
    status("Memulai proses scanning...")
    scan_id, result = start_scan(repo_url)
    if not scan_id:
        print()
        status("Gagal mendapatkan hasil scan.", "!")
        return
    print()
    status(f"Scan ID : {scan_id}", "+")
    status("Memproses hasil scan...")
    show_result(result)

def run():
    while True:
        try:
            main()
            print()
            print("  " + "─" * 58)
            print()
            choice = input("  Scan repository lain? [y/N] > ").strip().lower()
            if choice != "y":
                print()
                status("Program selesai. Sampai jumpa.", "+")
                break
            clear_screen()
            banner()
        except KeyboardInterrupt:
            print()
            status("Program dihentikan oleh user.", "!")
            break
        except Exception as error:
            print()
            status(f"Error: {error}", "!")
            break

if __name__ == "__main__":
    run()

const BASE_URL = "https://api-router.kaspersky-labs.com/whocalls";

const DEFAULT_HEADERS = {
  "sec-ch-ua-platform": "\"Android\"",
  "user-agent": "Mozilla/5.0 (Linux; Android 10; K) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/151.0.0.0 Mobile Safari/537.36",

  "sec-fetch-site": "cross-site",
  "sec-fetch-mode": "cors",
  "sec-fetch-dest": "empty",
  "referer": "https://whocalls.id.kaspersky.com/",
  "accept-encoding": "gzip, deflate, br, zstd",
  "accept-language": "en-US,en;q=0.9,id-ID;q=0.8,id;q=0.7",
  "priority": "u=1, i"
};

function buildQuery(params) {
  return new URLSearchParams(params).toString();
}

function normalizeNomor(nomor) {
  let n = nomor.replace(/[^\d]/g, "");
  if (n.startsWith("0")) {
    n = "62" + n.slice(1);
  } else if (n.startsWith("620")) {
    n = "62" + n.slice(3);
  }
  return n;
}

async function apiGet(path, params, label) {
  const query = buildQuery(params);
  const url = BASE_URL + path + "?" + query;
  const res = await fetch(url, { method: "GET", headers: DEFAULT_HEADERS });
  const bodyText = await res.text();
  if (!res.ok) {
    throw new Error("[" + label + "] HTTP " + res.status + " - " + res.statusText + "\nURL: " + url + "\nResponse: " + bodyText);
  }
  try {
    return JSON.parse(bodyText);
  } catch (e) {
    throw new Error("[" + label + "] Response bukan JSON valid: " + bodyText);
  }
}

async function cekInfoNomor(nomor, language) {
  return apiGet("/v2/number", { language: language || "id", number: nomor }, "cekInfoNomor");
}

async function cariNomorSerupa(nomor, region) {
  return apiGet("/similar-numbers", { region: region || "ID", number: nomor }, "cariNomorSerupa");
}

function parseArgs(argv) {
  const args = { nomor: null, lang: "id", region: "ID", raw: false };
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i];
    if (a === "--no" || a === "--nomor") args.nomor = argv[++i];
    else if (a === "--lang") args.lang = argv[++i];
    else if (a === "--region") args.region = argv[++i];
    else if (a === "--raw") args.raw = true;
  }
  return args;
}

async function main() {
  const argv = process.argv.slice(2);
  const parsed = parseArgs(argv);
  const nomor = parsed.nomor;
  const lang = parsed.lang;
  const region = parsed.region;
  const raw = parsed.raw;

  if (!nomor) {
    console.error("Penggunaan: node cek.js --no  [--lang id] [--region ID] [--raw]");
    process.exit(1);
  }

  const target = raw ? nomor : normalizeNomor(nomor);
  console.log("Nomor input : " + nomor);
  console.log("Nomor dipakai: " + target + "\n");

  try {
    const info = await cekInfoNomor(target, lang);
    console.log("=== Info Nomor ===");
    console.log(JSON.stringify(info, null, 2));
  } catch (err) {
    console.error("Gagal ambil info nomor:\n" + err.message);
  }

  try {
    const mirip = await cariNomorSerupa(target, region);
    console.log("\n=== Nomor Serupa ===");
    console.log(JSON.stringify(mirip, null, 2));
  } catch (err) {
    console.error("\nGagal ambil nomor serupa:\n" + err.message);
  }
}

main();
